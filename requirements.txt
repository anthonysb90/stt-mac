# --- transcription engines, selected per architecture -----------------------
# Environment markers do the picking, so a single `pip install -r` is correct
# on both machines.

# Apple Silicon: Parakeet on MLX. MLX has no x86_64 build, so this is arm64
# only. Needs Python 3.10+ and ffmpeg on PATH.
parakeet-mlx>=0.5.0 ; sys_platform == "darwin" and platform_machine == "arm64"

# Intel: faster-whisper (CTranslate2), int8 on CPU. Also installable on Apple
# Silicon by hand if you want a second opinion on a transcript.
faster-whisper>=1.1.0 ; sys_platform == "darwin" and platform_machine == "x86_64"
